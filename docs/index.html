// missing years as explicit gaps (y = NaN; Plot would draw null as 0), so lines break instead of bridging years with no report
withGaps = (data, y, z) => {
  const groups = z ? d3.groups(data, d => d[z]) : [[null, data]]
  return groups.flatMap(([k, v]) => {
    const have = new Map(v.filter(d => d[y] != null).map(d => [d.FY, d]))
    return d3.range(lo, hi + 1).map(FY => have.get(FY) || Object.assign({FY, [y]: NaN}, z ? {[z]: k} : {}))
  })
}
// a measure over the years: lines and dots; with a single year, horizontal bars from zero
// (one dot on a zoomed-in axis would exaggerate tiny differences)
function trend({data, y, z = null, names = null, colors = [BRAND], yopt = {}, title, dashed = null, rules = [], clip = false, height = CH}) {
  const color = names ? {domain: names, range: colors, legend: true} : undefined
  if (lo === hi) {
    const nm = names || [yopt.label || "Value"]
    const rows = data.filter(d => d.FY === hi && d[y] != null)
    if (!rows.length) return note(`Nothing reported for FY${hi}.`)
    return Plot.plot({height: Math.max(120, 60 + 38 * nm.length), width: PAIR, marginLeft: Math.min(220, 16 + 6.5 * d3.max(nm, n => n.length)),
      x: {grid: true, label: `FY${hi}${yopt.label ? ": " + yopt.label : ""}`, percent: !!yopt.percent, tickFormat: yopt.tickFormat},
      y: {domain: nm, label: null}, color,
      marks: [Plot.barX(rows, {x: y, y: z || (() => nm[0]), fill: z || colors[0], tip: true, title}), Plot.ruleX([0])]})
  }
  // two years: a line would be one straight segment, so bars side by side (the axis starts at zero, like every bar chart here)
  if (hi - lo === 1) {
    const nm = names || [yopt.label || "Value"]
    const rows = data.filter(d => d[y] != null && !Number.isNaN(d[y]))
    if (!rows.length) return note(`Nothing reported for FY${lo}–FY${hi}.`)
    return Plot.plot({height, width: PAIR, marginLeft: 60, fx: TWO_YEARS, x: {axis: null, domain: nm}, y: Object.assign({grid: true, nice: true}, yopt), color,
      marks: [...rules, Plot.barY(rows, {fx: "FY", x: z || (() => nm[0]), y, fill: z || colors[0], tip: true, title, clip}), Plot.ruleY([0])]})
  }
  const known = data.filter(d => d[y] != null && !Number.isNaN(d[y]))
  if (!known.length) return note(`Nothing reported for FY${lo}–FY${hi}.`)
  const solid = withGaps(dashed ? data.filter(d => !dashed(d)) : data, y, z)
  return Plot.plot({height, width: PAIR, marginLeft: 60, x: yearX, y: Object.assign({grid: true}, yopt), color,
    marks: [...rules,
            Plot.lineY(solid, {x: "FY", y, stroke: z || colors[0], strokeWidth: 2, clip}),
            ...(dashed ? [Plot.lineY(withGaps(data.filter(dashed), y, z), {x: "FY", y, stroke: z || colors[0], strokeWidth: 2, strokeDasharray: "5,3", clip})] : []),
            Plot.dot(known, {x: "FY", y, fill: z || colors[0], r: 3, tip: true, title, clip})]})
}

function growth(rows, f) {
  const key = d => `${d.level}|${d.Agency}|${d.Key}`
  const by = d3.group(rows.filter(d => f(d) != null), d => d.FY)
  return byYear([...by].filter(([FY]) => by.has(FY - 1)).map(([FY, v]) => {
    const prev = new Map(by.get(FY - 1).map(d => [key(d), f(d)]))
    const both = v.filter(d => prev.has(key(d)))
    const before = d3.sum(both, d => prev.get(key(d)))
    return Object({FY, growth: before ? d3.sum(both, f) / before - 1 : null, n: both.length})
  }))
}

function glance(q) {
  const at = FY => ROWS.filter(d => d.FY === FY && inSel(d))

    if (merits == null) return all == null ? "" : "no appeals were decided on the merits"
    return `${p(merits)} of appeals decided on the merits were reversed fully or partly, which is ${p(all)} of all processed appeals`
  }
  if (q === 6) {

  }
  return ""
}
